01

DDL — Data Definition Language

Veri tabanını oluşturma yapısını tanımlama, yapısını değiştirme ve silme işlemlerini içerir.

Komutlar:  CREATE ALTER DROP TRUNCATE

CREATE Komutu

Veri tabanı veya tablo oluşturmada kullanılır.

Create ile Veri Tabanı Oluşturma

SQL
CREATE DATABASE Siparisler;

Create ile Tablo Oluşturma

SQL
CREATE TABLE Ürünler (
    ürünID   int Primary Key,
    ürünAd   int,
    ürünAdet int,
);

ALTER Komutu

Mevcut bir tablo üzerinde değişiklikler yapmak için kullanılır. Tabloya sütun eklenebilir, var olan sütun değiştirilebilir veya silinebilir.

Alter ile Tabloya Sütun Ekleme

SQL
ALTER TABLE Ürünler ADD eklenecekSütun int;

Alter ve Drop ile Tablodan Sütun Kaldırma

SQL
ALTER TABLE Ürünler DROP COLUMN silinecekSütun;

Alter ile Sütunun Veri Tipini Değiştirme

SQL
ALTER TABLE Ürünler ALTER COLUMN değişecekSütun int;

Alter ile Sütun Adını Değiştirme

SQL
EXEC sp_rename 'Ürünler.eskiSütun', 'yeniSütun', 'COLUMN';

Alter ile Tablo İsmini Değiştirme

SQL
EXEC sp_rename 'Ürünler', 'NewÜrünler';

DROP Komutu

Veritabanı veya tablo silme komutudur.

Drop ile Veritabanını Silme

SQL
DROP DATABASE Siparisler;

Drop ile Tablo Silme

SQL
DROP TABLE Ürünler;

Alter ve Drop ile Tablodan Sütun Silme

SQL
ALTER TABLE Ürünler DROP COLUMN silinecekSütun;

TRUNCATE Komutu

Truncate ile Tablodaki Verileri Silme

SQL
TRUNCATE TABLE Ürünler;
02

DML — SELECT

Veritabanındaki verileri sorgulamak, getirmek, düzenlemek ve özetlemek için kullanılır. En sık kullanılan komutlar: SELECT, INSERT, DELETE, UPDATE

SELECT: Tek tablo içinde çeşitli kriterlere göre sorgulama, sıralama, özetleme ve ortalama gibi işlemler yapmak için kullanılır. Gerekli bilgileri öne çıkarmak amacıyla kullanılır.

Tablodaki tüm verileri getirir

SQL
SELECT * FROM Ürünler;

Ürünler tablosundan ürün adlarını getirir

SQL
SELECT ürünAd FROM Ürünler;

Birden fazla sütun getirme — Müşteriler tablosundan müşteri ad ve soyadları getirir

SQL
SELECT müsteriAD, müsteriSoyad FROM Müsteriler;

DISTINCT — Aynı Olanlardan Sadece Birini Getirir

SQL
SELECT DISTINCT sehir FROM Müsteriler;
-- Müsteriler tablosunda sehir sütununda iki tane "İstanbul" varsa sadece bir tanesini gösterir.

WHERE — Koşula Göre Veri Getirme

SQL
SELECT * FROM Ögrenciler WHERE vizeNotu > 60;
-- Öğrencilerin kayıtlarından vize notu 60'tan büyük olanları getirir.

SELECT * FROM Ögrenciler WHERE ögrenciAd = 'Umut';
-- Öğrencilerden adı Umut olanları getirir.

AND — Birden Fazla Koşul Sağlandığında

SQL
SELECT * FROM Ögrenciler WHERE vizeNotu > 60 AND ögrenciAd = 'Umut';
-- Öğrencilerden vize notu 60'tan büyük ve ismi Umut olanları getirir.

OR

SQL
SELECT * FROM Ögrenciler WHERE vizeNotu > 60 OR ögrenciAd = 'Umut';
-- Öğrencilerden vize notu 60'tan büyük olanları veya ismi Umut olanları getirir

BETWEEN — İki Değer Arası

SQL
SELECT * FROM tabloAdı WHERE sütunAdı1 BETWEEN 10 AND 15;
-- 10 ile 15 arasındaki değerleri getirir (10 ve 15 dahil).

NOT — Koşulun Tersi

SQL
SELECT * FROM Ögrenciler WHERE NOT ögrenciAd = 'Umut'
-- Ögrencilerden ismi "Umut" olmayanları getirir.

LIKE — Karakter Türü Verilerde Arama

SQL
SELECT * FROM Ögrenciler WHERE ögrenciAd LIKE 'U%'
-- U harfiyle başlayan ögrencileri getirir.

SELECT * FROM Ögrenciler WHERE ögrenciAd LIKE '%UMU%'
-- İçinde UMU geçen öğrenci kayıtlarını getirir.

SELECT * FROM Ögrenciler WHERE ögrenciAd NOT LIKE '%U%'
-- İçinde U harfi geçmeyen Ögrenci kayıtlarını getirir

IN, NOT IN — Belirli Bir Küme İçinde Arama (OR gibidir aslında)

SQL
SELECT * FROM tabloAdı WHERE sütunAdı IN (7, 12)
-- Değeri 7 veya 12 olan kayıtları getirir.

SELECT * FROM Ögrenciler WHERE ögrenciAd IN ('Umut', 'Zeynep', 'Mehmet') OR ögrenciAd LIKE 'A%'
-- Adı Umut, Zeynep, Mehmet olanları veya Adı A ile başlayanları getirir

SELECT * FROM Ögrenciler WHERE ögrenciAd NOT IN ('Umut')
-- Öğrencilerden adı Umut olmayanları getirir

ORDER BY — Sıralama

SQL
SELECT * FROM Ürünler ORDER BY stok_miktari
-- Stok sayısı küçükten büyüğe sıralanır

SELECT * FROM Müsteriler WHERE müsteriID>5 AND müsteriID<9 ORDER BY müsteriAd
-- Müsteri ID'si 6 7 8 olanların isimlerini alfabetik sıralar

-- ASC  → Artan sırada (varsayılan)
-- DESC → Azalan sırada

-- Örnek: OgrAd, OgrID desc denirse eğer, burada Adlar alfabetik sıralanır
-- aynı isimde olanların ise ID'si büyükten küçüğe sıralanır.

AS — Geçici Sütun

SQL
SELECT (Vize+Final)/2 AS 'Ortalama' FROM Notlar
-- Vize Final notlarının ortalamasını alıp geçici oluşturulan Ortalama sütununda gösterir.

SELECT Vize, Vize+5 AS '+5 Eklenmiş Hali' FROM Notlar
-- Vize notlarının normalini bir de +5 eklediğimiz +5 Eklenmiş Hali sütununu gösterir.

GROUP BY — COUNT

SQL
SELECT kitapTür, COUNT(kitapAdi) AS 'Kitap Sayisi' FROM Kitaplar GROUP BY kitapTür
-- Kitapların türlerinden kaç tane var bulur ve bunu Kitap Sayisi sütununda tutar.

SELECT kitapAdi, COUNT(kitapAdi) AS 'Kitap Sayısı' FROM Kitaplar GROUP BY kitapAdi
-- Kitapların adlarına göre kaç tane olduğunu bulur ve bunu Kitap Sayisi sütununda tutar.

SELECT departmanAdi, COUNT(departmanID) FROM Departmanlar GROUP BY departmanAdi
-- Departman Adı aynı olanları gruplandırır ve bunların sayısını alır.

SELECT COUNT(kitapAdi) FROM Kitaplar WHERE Fiyat>45;
-- Kitapların fiyatının 45 üzeri olanların sayısını getirir.

SELECT COUNT(DISTINCT kitapFiyat) FROM Kitaplar;
-- Kitapların fiyatlarından aynı olanları çıkarır da öyle sayar.

SELECT ürünKategori, COUNT(*) AS 'Ürünlerin Sayısı' FROM Ürünler GROUP BY ürünKategori
-- Ürünün kategorisi ve adeti görünür ekranda.

SELECT COUNT(*) AS 'Ürünlerin Sayısı' FROM Ürünler GROUP BY ürünKategori
-- Burada ürün kategorilere ayrılır ama sadece bunun sayısı gösterilir. 2 2 3 gibi mesela.
-- Ama 2 dediği 3 dediği hangi kategori bilinmez.

HAVING — GROUP BY'ın Şartıdır

SQL
SELECT müsteriSehir, COUNT(*) FROM Müsteriler
GROUP BY müsteriSehir
HAVING müsteriSehir != 'Ankara'
-- Ankara hariç diğer şehirleri gruplar ve bunları şehirlere göre gösterir.

Kendi benzetmem: Group by'dan sonra gelen kısım sanki backend yazılımcısıymış gibi bir şey — zira arkada o kategorileri ayıran kişi o aslında. Sadece ekrana getirmiyor. Bunu ekrana getiren frontendci yani SELECT'ten sonra gelen. Onu yazmasaydık ekranda sadece ayrılan kategori sayılarını görürdük.

Matematiksel Operatörler

SUM

SQL
SELECT SUM(ürünStok) FROM Ürünler
-- Ürün tablosundaki ürünlerin toplam stok adedini bulur.

SELECT SUM(ürünStok) FROM Ürünler WHERE ürünFiyat> 20
-- Ürün fiyatı 20'den büyük olanların toplam stok adedini yazdırır.

SELECT ürünAdi, SUM(ürünFiyat * ürünStok) AS 'TOPLAM' FROM Ürünler GROUP BY ürünAdi
-- Ürünlerin toplam fiyatını TOPLAM sütunu altında tutar.

SELECT ürünKategori, SUM(ürünStok) FROM Ürünler GROUP BY ürünKategori
-- Ürünleri kategorilere ayırıp bunların toplam stok miktarlarını bulur

SELECT SUM(ürünStok) AS 'Toplam Silgi, Dosya, Makas' FROM Ürünler
WHERE ürünAdı IN ('Makas', 'Silgi', 'Dosya')
-- Ürün adları makas, silgi ve dosya olanların toplam stok adedini verir.

MAX — MIN

SQL
SELECT MAX(kitapFiyat) FROM Kitaplar
-- Kitapların fiyatından en büyük olanı getirir.

SELECT MIN(ürünFiyat) FROM Ürünler WHERE ürünKategori = 'Kalem'
-- Kategorisi kalem olanlardan en düşük fiyatlı olanı getirir

SELECT MAX(ürünFiyat) AS 'En Yüksek Fiyat', MIN(ürünFiyat) AS 'En Düşük Fiyat' FROM Ürünler
-- Ürünler tablosundaki en yüksek satış fiyatı ile en düşük satış fiyatını listeler

AVG

SQL
SELECT AVG(ürünStok) FROM Ürünler
-- Toplam stok miktarının ortalamasını alır.

SELECT * FROM Ürünler WHERE ürünFiyat > (SELECT AVG(ürünFiyat) FROM Ürünler)
-- Fiyatları, ürünlerin fiyat ortalamasından büyük olan ürünleri yazdırır.

Metinsel Fonksiyonlar

ASCII — Karakterin ASCII Kodunu Verir

SQL
SELECT ASCII('A')
-- 65 çıktısını verir

CHAR — ASCII Koduna Karşılık Gelen Harfi Verir

SQL
SELECT CHAR(65)
-- A harfini verir.

CHARINDEX — Kelimenin Kaçıncı İndeksten Başladığını Gösterir

SQL
SELECT CHARINDEX('merhaba', 'hello merhaba dünya')
-- merhaba kelimesinin kaçıncı indexten başladığını gösterir. Bu komutta 7.indexten başlar.

LEFT — Soldan Belirli Sayıda Karakter Alır

SQL
SELECT LEFT('Merhaba Dünya', 3)
-- Soldan 3 indexi alır. Çıktısı Mer olur

LEN — Karakter Sayısını Verir

SQL
SELECT LEN('Merhaba Dünya')
-- Kaç karakter olduğu bilgisini verir. Çıktısı 13 olur

SELECT LEN(urunAdi) FROM tbl_urunler

UPPER — LOWER

SQL
SELECT UPPER(urun_adi) FROM tbl_urunler
-- Ürünler tablosundaki ürün adlarını büyük harfe çevirir

SELECT LOWER(urun_adi) FROM tbl_urunler
-- Ürünler tablosundaki ürün adlarını küçük harfe çevirir.

REPLICATE

SQL
SELECT REPLICATE(urun_adi, 2) FROM tbl_urunler
-- Ürünler tablosundaki ürün adlarını 2 defa yazdırır

LTRIM — RTRIM

SQL
SELECT LTRIM(ad) FROM tbl_personeller
-- Soldaki boşlukları siler

SELECT RTRIM(ad) FROM tbl_personeller
-- Sağdaki boşlukları siler.

REVERSE

SQL
SELECT REVERSE(ad) FROM tbl_personeller
-- Adları tersten yazar.

SUBSTRING

SQL
SELECT SUBSTRING('Merhaba', başlangıç, uzunluk)
SELECT SUBSTRING('Merhaba', 3, 5)
-- Belli bir kısmı alır. Index SQL'de 1'den başlar.
-- Çıktısı: rhaba  3.indexle başlayıp 5 karakter sayılır, 3.index dahil

Birleştirmeler (JOIN)

JOIN Türleri:  INNER JOIN LEFT JOIN RIGHT JOIN FULL OUTER JOIN

INNER JOIN

İki tablodaki ortak alanları birleştirmede kullanılır.

SQL
SELECT * FROM tbl_personeller
INNER JOIN tbl_departman
ON tbl_personeller.departman_id = tbl_departman.departman_id
-- Personeller tablosu ile departman tablosunu birleştir.
-- Birleştirirken, departman_id'si aynı olan kayıtları eşleştirerek birleştir.

SELECT ad, tbl_departman.departman_id
FROM tbl_personeller
INNER JOIN tbl_departman
ON tbl_personeller.departman_id = tbl_departman.departman_id
-- Üstteki örneğin sadece ad ve departman id kısımlarını getiren halidir.
-- Departman ID iki yerde olduğundan konum belirtilmelidir.

LEFT JOIN

SQL
SELECT * FROM tbl_personeller
LEFT JOIN tbl_departman
ON tbl_personeller.departman_id = tbl_departman.departman_id
-- Soldaki tablonun tamamı, sağdaki tablodan sadece eşleşenler gelir

-- Örnek çıktı:
-- ad       departman_adi
-- Ali      IT
-- Ayşe     Muhasebe
-- Mehmet   NULL

RIGHT JOIN

SQL
SELECT * FROM tbl_personeller
RIGHT JOIN tbl_departman
ON tbl_personeller.departman_id = tbl_departman.departman_id
-- Sağdaki tablonun tamamı, soldaki tablodan sadece eşleşenler gelir

-- Örnek çıktı:
-- ad       departman_adi
-- Ali      IT
-- Ayşe     Muhasebe
-- NULL     İnsan Kaynakları

FULL OUTER JOIN

Hem sağdaki hem soldaki tabloya göre birleştirme yapar.

SQL
SELECT * FROM tbl_personeller
FULL OUTER JOIN tbl_departman
ON tbl_personeller.departman_id = tbl_departman.departman_id

-- Örnek çıktı:
-- ad       departman_adi
-- Ali      IT
-- Ayşe     Muhasebe
-- Mehmet   NULL
-- NULL     İnsan Kaynakları

T-SQL

Değişken tanımlamak için DECLARE komutu kullanırız.

Değişken Tanımlama ve Kullanım

Ad isminde bir değişken oluşturup atama yapılıyor ve sonra çağrılıyor.

SQL
DECLARE @ad varchar(20)
SET @ad = 'Ümüdo'
SELECT @ad

Değişken olarak tanımlanan iki sayıyı toplayan sorgu

SQL
DECLARE @sayı1 int, @sayı2 int, @sonuc int
SET @sayı1 = 15
SET @sayı2 = 50
SET @sonuc = @sayı1 + @sayı2
SELECT @sonuc AS 'TOPLAM'

Değişken olarak tanımlanan üç sayının ortalamasını alan sorgu

SQL
DECLARE @sayi1 int, @sayi2 int, @sayi3 int, @ortalama float;
SET @sayi1 = 10
SET @sayi2 = 20
SET @sayi3 = 30
SET @ortalama = (@sayi1 + @sayi2 + @sayi3) / 3.0
SELECT @ortalama AS 'Ortalama'

Değişken olarak tanımlanan 200 sayısının %15'ini alan sorgu

SQL
DECLARE @sayi float, @yuzde float, @sonuc float
SET @sayi = 200
SET @yuzde = 15
SET @sonuc = @sayi * (@yuzde / 100)
SELECT @sonuc AS 'Yüzde 15'

WHILE

10'a kadar olan sayıları toplayan sorgu

SQL
DECLARE @sayi int = 1;
DECLARE @toplam int = 0;

WHILE @sayi <= 10
BEGIN
    SET @toplam = @toplam + @sayi;
    SET @sayi = @sayi + 1;
END

Örnek 1: Stok değerine 20 ekleyen fonksiyon

SQL
CREATE FUNCTION StokArtir (@stok INT)
RETURNS INT
AS
BEGIN
    RETURN @stok + 20;
END;

SELECT urun_adi, stok_miktari, dbo.StokArtir(stok_miktari) FROM tbl_urunler;

Örnek 2: Çalışma süresi hesaplayan fonksiyon (DateDiff)

SQL
CREATE FUNCTION CalismaSuresi (@iseBaslama DATE)
RETURNS INT
AS
BEGIN
    RETURN DATEDIFF(YEAR, @iseBaslama, GETDATE());
END;

SELECT ad, soyad, ise_baslama_tarihi,
       dbo.CalismaSuresi(ise_baslama_tarihi) AS CalismaYili
FROM tbl_personeller;

Trigger (Tetikleyiciler)

INSERT, UPDATE, DELETE işlemlerinden sonra otomatik çalışan kod bloklarıdır.

1. Insert Trigger: Ürün eklenince mesaj ver

SQL
CREATE TRIGGER Urunekle
ON tbl_urunler
AFTER INSERT
AS
BEGIN
    SELECT 'Yeni bir ürün eklendi';
END;

INSERT INTO tbl_urunler VALUES ('Dolma Kalem', 'Kırtasiye', 10, 12.3)

2. Insert Trigger: Toplam stok miktarını göster

SQL
CREATE TRIGGER Urunekle2
ON tbl_urunler
AFTER INSERT
AS
BEGIN
    SELECT SUM(stok_miktari) FROM tbl_urunler;
END;

INSERT INTO tbl_urunler VALUES ('Hesap Makinesi', 'Kırtasiye', 20, 12.3)

3. Update Trigger: Güncelleme mesajı

SQL
CREATE TRIGGER GuncellemeBilgisi
ON tbl_urunler
AFTER UPDATE
AS
BEGIN
    SELECT 'Güncelleme işlemi gerçekleşti';
END;

UPDATE tbl_urunler SET birim_fiyat = birim_fiyat - 2 WHERE urun_id = 2

4. Update Trigger: Güncelleme sayacını artır (Başka tabloyu etkiler)

SQL
CREATE TRIGGER GuncellemeKontrol
ON tbl_urunler
AFTER UPDATE
AS

    UPDATE tbl_guncelleme SET sayac = sayac + 1;
END;

5. Delete Trigger: Silme mesajı

SQL
CREATE TRIGGER SilmeKontrol
ON tbl_urunler
AFTER DELETE
AS

    SELECT 'Kayıt tablodan silindi';
END;

SQL

ON tbl_urunler

Bir tablonun birincil anahtarı başka tabloda ikincil anahtar olabilir → tablolar arası bağlantı kurar.
